cmake_minimum_required(VERSION 3.16)
project(qidian_script_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(../scripts scripts)
set(sources script_test.cpp ../scripts/script_runtime.cpp ../scripts/script_crypto.cpp ../primitives/default_signature.cpp ../primitives/fock_digest.cpp)
if(ANDROID)
  add_library(qidian_script_tests SHARED ${sources})
  target_link_libraries(qidian_script_tests PRIVATE m)
else()
  add_executable(qidian_script_tests ${sources})
  enable_testing()
  add_test(NAME qidian_script_tests COMMAND qidian_script_tests)
endif()
target_link_libraries(qidian_script_tests PRIVATE qidian_script_engine qidian_script_crypto)
target_include_directories(qidian_script_tests PRIVATE ../scripts ../primitives)
