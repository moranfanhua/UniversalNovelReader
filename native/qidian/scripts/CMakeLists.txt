set(qjs_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/qidian_quickjs")
set(crypto_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/qidian_mbedtls")
add_library(qidian_script_engine STATIC "${qjs_root}/quickjs.c" "${qjs_root}/libregexp.c" "${qjs_root}/libunicode.c" "${qjs_root}/cutils.c")
set_target_properties(qidian_script_engine PROPERTIES C_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(qidian_script_engine PRIVATE CONFIG_VERSION="2021-03-27" _GNU_SOURCE)
target_include_directories(qidian_script_engine SYSTEM PUBLIC "${qjs_root}")
if(MSVC)
  target_compile_options(qidian_script_engine PRIVATE /w /utf-8 "/FI${qjs_root}/compat/qjs_msvc.h")
  target_include_directories(qidian_script_engine PRIVATE "${qjs_root}/compat")
else()
  target_compile_options(qidian_script_engine PRIVATE -w)
endif()
add_library(qidian_script_crypto STATIC "${crypto_root}/library/aes.c" "${crypto_root}/library/des.c" "${crypto_root}/library/blowfish.c" "${crypto_root}/library/camellia.c" "${crypto_root}/library/xtea.c" "${crypto_root}/library/platform_util.c")
set_target_properties(qidian_script_crypto PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(qidian_script_crypto PUBLIC "${crypto_root}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(qidian_script_crypto PUBLIC MBEDTLS_CONFIG_FILE="crypto_config.h")
